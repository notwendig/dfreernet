cmake_minimum_required(VERSION 3.21)
if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
  project(qt_attendant VERSION 1.5.1 LANGUAGES CXX)
  set(CMAKE_CXX_STANDARD 17)
  set(CMAKE_CXX_STANDARD_REQUIRED ON)
  set(CMAKE_CXX_EXTENSIONS OFF)
  include(CTest)
endif()
if(NOT CMAKE_SYSTEM_NAME STREQUAL "Linux")
  message(FATAL_ERROR "qt_attendant requires Linux SocketCAN")
endif()
find_package(Qt6 6.10 REQUIRED COMPONENTS Core Gui Widgets Quick QuickControls2 LinguistTools)
qt_policy(SET QTP0004 NEW)
add_library(attendant_core STATIC startup_ready.cpp startup_tables.cpp module_enrollment.cpp startup_config.cpp registered_resume.cpp can_control.cpp capture_playback.cpp attendant_state.cpp)
target_include_directories(attendant_core PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}")
target_link_libraries(attendant_core PUBLIC Qt6::Core)
target_compile_options(attendant_core PRIVATE -Wall -Wextra -Wpedantic)
add_library(attendant_runtime STATIC can_observer.cpp can_transmitter.cpp attendant_controller.cpp)
set_target_properties(attendant_runtime PROPERTIES AUTOMOC ON)
target_link_libraries(attendant_runtime PUBLIC attendant_core Qt6::Gui)
target_compile_options(attendant_runtime PRIVATE -Wall -Wextra -Wpedantic)
add_library(attendant_tray STATIC tray_controller.cpp app_icon.cpp)
set_target_properties(attendant_tray PROPERTIES AUTOMOC ON)
target_link_libraries(attendant_tray PUBLIC attendant_runtime Qt6::Widgets Qt6::Quick)
target_compile_options(attendant_tray PRIVATE -Wall -Wextra -Wpedantic)
qt_add_executable(qt_attendant main.cpp shutdown_signals.cpp)
target_compile_definitions(qt_attendant PRIVATE ATTENDANT_VERSION="${PROJECT_VERSION}")
set_target_properties(qt_attendant PROPERTIES AUTOMOC ON)
target_link_libraries(qt_attendant PRIVATE attendant_tray Qt6::QuickControls2)
target_compile_options(qt_attendant PRIVATE -Wall -Wextra -Wpedantic)
qt_add_qml_module(qt_attendant URI Dfree.Attendant VERSION 1.0 RESOURCE_PREFIX /qt/qml
  QML_FILES qml/CapturePlayback.qml qml/Main.qml qml/JoystickPad.qml)
qt_add_translation(ATTENDANT_QM "${CMAKE_CURRENT_SOURCE_DIR}/../translations/rnet_de_DE.ts")
qt_add_resources(qt_attendant attendant_translations PREFIX /i18n BASE "${CMAKE_CURRENT_BINARY_DIR}" FILES ${ATTENDANT_QM})
set_target_properties(qt_attendant PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}")
install(TARGETS qt_attendant RUNTIME DESTINATION bin)
install(FILES dfreernet-attendant.desktop DESTINATION share/applications)
install(FILES icons/dfreernet-attendant.svg DESTINATION share/icons/hicolor/scalable/apps)
install(FILES README.md README.de.md DESTINATION share/doc/dfreernet/qt_attendant)
install(FILES "${CMAKE_CURRENT_SOURCE_DIR}/../docs/screens/attendant-safe-de.png"
        "${CMAKE_CURRENT_SOURCE_DIR}/../docs/screens/attendant-tray-de.png"
        "${CMAKE_CURRENT_SOURCE_DIR}/../docs/screens/attendant-power-on-wait-de.png"
        "${CMAKE_CURRENT_SOURCE_DIR}/../docs/screens/attendant-registered-resume-de.png"
        DESTINATION share/doc/dfreernet/docs/screens OPTIONAL)
if(BUILD_TESTING)

  set_tests_properties(attendant_can_setup PROPERTIES TIMEOUT 20)

  target_link_libraries(attendant_quick_test PRIVATE attendant_runtime Qt6::QuickTest Qt6::QuickControls2)
  add_test(NAME attendant_touch_ui COMMAND attendant_quick_test)
  set_tests_properties(attendant_touch_ui PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen;QT_QUICK_BACKEND=software" TIMEOUT 30)
