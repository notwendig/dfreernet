cmake_minimum_required(VERSION 3.21)
project(dfreernet_wine LANGUAGES CXX)
include(CTest)
if(WIN32)
  if(NOT CMAKE_SIZEOF_VOID_P EQUAL 4)
    message(FATAL_ERROR "Use cmake/mingw32.cmake: the Programmer requires PE32")
  endif()
  add_library(ftd2xx SHARED ftd2xx/ftd2xx_emu.cpp ftd2xx/ftd2xx.def)
  set_target_properties(ftd2xx PROPERTIES PREFIX "" OUTPUT_NAME ftd2xx)
  target_include_directories(ftd2xx PRIVATE common)
  target_compile_definitions(ftd2xx PRIVATE WIN32_LEAN_AND_MEAN NOMINMAX)
  target_link_options(ftd2xx PRIVATE -static)
  add_executable(wine_dll_test tests/dll_test.cpp common/DongleMessages.cpp)
  target_include_directories(wine_dll_test PRIVATE common)
  target_link_options(wine_dll_test PRIVATE -static)
  add_executable(wine_factory_limit_test tests/factory_limit_test.cpp)
  target_link_options(wine_factory_limit_test PRIVATE -static)
  foreach(target ftd2xx wine_dll_test wine_factory_limit_test)
    target_compile_features(${target} PRIVATE cxx_std_17)
    target_compile_options(${target} PRIVATE -Wall -Wextra -Wpedantic)
  endforeach()
else()
  add_executable(dfreernet-wine-can-proxy proxy/main.cpp)
  target_compile_features(dfreernet-wine-can-proxy PRIVATE cxx_std_17)
  target_compile_options(dfreernet-wine-can-proxy PRIVATE -Wall -Wextra -Wpedantic)
  if(BUILD_TESTING)
    add_executable(wine_messages_test tests/messages_test.cpp common/DongleMessages.cpp)
    target_include_directories(wine_messages_test PRIVATE common)
    target_compile_features(wine_messages_test PRIVATE cxx_std_17)
    add_test(NAME wine_message_conversion COMMAND wine_messages_test)
    find_package(Python3 REQUIRED COMPONENTS Interpreter)
    add_test(NAME wine_factory_patch_guards COMMAND "${Python3_EXECUTABLE}"
      "${CMAKE_CURRENT_SOURCE_DIR}/tests/factory_patch_test.py")
  endif()
endif()
