cmake_minimum_required(VERSION 3.21)
if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
  project(qt_iredit VERSION 1.5.1 LANGUAGES CXX)
  set(CMAKE_CXX_STANDARD 17)
  set(CMAKE_CXX_STANDARD_REQUIRED ON)
  set(CMAKE_CXX_EXTENSIONS OFF)
  include(CTest)
  find_package(Qt6 6.10 REQUIRED COMPONENTS Core Widgets LinguistTools)
endif()
add_library(iredit_core STATIC ir_document.cpp)
target_include_directories(iredit_core PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}")
target_link_libraries(iredit_core PUBLIC Qt6::Core)
target_compile_options(iredit_core PRIVATE -Wall -Wextra -Wpedantic)
add_library(iredit_ui STATIC editor_window.cpp)
target_link_libraries(iredit_ui PUBLIC iredit_core Qt6::Widgets)
target_compile_options(iredit_ui PRIVATE -Wall -Wextra -Wpedantic)
qt_add_translation(IREdit_QM "${CMAKE_CURRENT_SOURCE_DIR}/../translations/rnet_de_DE.ts")
add_executable(qt_iredit main.cpp)
target_link_libraries(qt_iredit PRIVATE iredit_ui)
target_compile_definitions(qt_iredit PRIVATE QT_IREDIT_VERSION="${PROJECT_VERSION}")
target_compile_options(qt_iredit PRIVATE -Wall -Wextra -Wpedantic)
qt_add_resources(qt_iredit iredit_translations PREFIX "/i18n" BASE "${CMAKE_CURRENT_BINARY_DIR}" FILES ${IREdit_QM})
set_target_properties(qt_iredit PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}")
install(TARGETS qt_iredit RUNTIME DESTINATION bin)
install(FILES README.md README.de.md DESTINATION share/doc/dfreernet/qt_iredit)
install(FILES "${CMAKE_CURRENT_SOURCE_DIR}/../docs/screens/ir-editor-0.1.0-en.png"
  "${CMAKE_CURRENT_SOURCE_DIR}/../docs/screens/ir-editor-0.1.0-de.png"
  DESTINATION share/doc/dfreernet/docs/screens)
if(BUILD_TESTING)
  find_package(Qt6 REQUIRED COMPONENTS Test)
  add_executable(qt_iredit_test "${CMAKE_CURRENT_SOURCE_DIR}/../tests/ir_editor_test.cpp")
  set_target_properties(qt_iredit_test PROPERTIES AUTOMOC ON)
  target_link_libraries(qt_iredit_test PRIVATE iredit_ui Qt6::Test)
  add_test(NAME qt_ir_editor COMMAND qt_iredit_test)
  set_tests_properties(qt_ir_editor PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen" TIMEOUT 30)
endif()
